--- Програма 1 (prog1_sort.py) ---
import re
import locale

# Встановлюємо українську локаль для правильного сортування алфавіту
try:
    locale.setlocale(locale.LC_ALL, 'uk_UA.UTF-8')
except locale.Error:
    try:
        locale.setlocale(locale.LC_ALL, 'Ukrainian_Ukraine.1251')
    except locale.Error:
        pass

def ukr_then_en_key(word):
    # Прибираємо розділові знаки для сортування
    clean_word = re.sub(r'[^a-zA-Zа-яА-ЯєЄіІїЇґҐ_]', '', word)
    first_char = clean_word[0] if clean_word else ''
    
    # Визначаємо, чи це кирилиця (українські літери)
    is_ukr = bool(re.match(r'[а-яА-ЯєЄіІїЇґҐ]', first_char))
    
    if is_ukr:
        # 0 = Першочергово українські слова
        return (0, locale.strxfrm(clean_word.lower()))
    else:
        # 1 = Латинські слова
        return (1, clean_word.lower())

--- Програма 2 (prog2_url.py) ---
import urllib.parse
import pyperclip

def decode_url(encoded_url: str) -> str:
    # Декодуємо URL з формату %D0%A8... в читабельну кирилицю
    decoded_url = urllib.parse.unquote(encoded_url)
    return decoded_url

def main():
    # Приклад URL із закодованою кирилицею
    encoded_url = "https://uk.wikipedia.org/wiki/%D0%A8%D1%82%D1%83%D1%87%D0%BD%D0%B8%D0%B9_%D1%96%D0%BD%D1%82%D0%B5%D0%BB%D0%B5%D0%BA%D1%82"
    
    print("Вхідне закодоване посилання:")
    print(encoded_url)
    
    decoded_url = decode_url(encoded_url)
    
    print("\nПеретворене посилання (з кирилицею):")
    print(decoded_url)
    
    # Копіюємо результат у буфер обміну
    try:
        pyperclip.copy(decoded_url)
        print("\nРезультат успішно скопійовано в буфер обміну!")
    except Exception as e:
        print(f"\nНе вдалося скопіювати в буфер обміну: {e}")

if __name__ == "__main__":
    main()

import json

def main():
    # Словник з 10 записів: {"Прізвище": ["Ім'я", "По батькові", рік народження]}
    data = {
        "Сніжко": ["Іван", "Богданович", 2004],
        "Шевченко": ["Тарас", "Григорович", 1814],
        "Франко": ["Іван", "Якович", 1856],
        "Українка": ["Леся", "Петрівна", 1871],
        "Коцюбинський": ["Михайло", "Михайлович", 1864],
        "Ліна": ["Костенко", "Василівна", 1930],
        "Стус": ["Василь", "Семенович", 1938],
        "Сковорода": ["Григорій", "Савич", 1722],
        "Нечуй-Левицький": ["Іван", "Семенович", 1838],
        "Рилський": ["Максим", "Тадейович", 1895]
    }

    json_filename = "students_data.json"

    # 1. Запис у JSON з ensure_ascii=False (збереження кирилиці як літер)
    with open(json_filename, "w", encoding="utf-8") as f:
        json.dump(data, f, ensure_ascii=False, indent=4)
    print(f"Дані успішно збережено у файл '{json_filename}' (кодування UTF-8).\n")

    # 2. Зчитування з JSON файлу та вивід у консоль
    with open(json_filename, "r", encoding="utf-8") as f:
        loaded_data = json.load(f)

    print("--- Прочитані дані з JSON файлу ---")
    for surname, info in loaded_data.items():
        print(f"{surname}: {info[0]} {info[1]}, {info[2]} р.н.")

if __name__ == "__main__":
    main()
